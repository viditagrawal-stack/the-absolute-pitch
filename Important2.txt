/* ============ SWERVE IT GAME ============ */
const cv = $('game'), cx = cv.getContext('2d');
let aimX = 210, shots, goals, ball, keeper, over;
const GOAL = { x: 110, w: 200, y: 40 }, WALL = { x: 150, w: 120, y: 250, h: 22 }, START = { x: 210, y: 400 };
function reset() { shots = 5; goals = 0; ball = null; keeper = { x: 210, dir: 1 }; over = false; $('gstat').textContent = 'Goals: 0 / Shots left: 5'; }
cv.addEventListener('mousemove', e => { const r = cv.getBoundingClientRect(); aimX = (e.clientX - r.left) * cv.width / r.width; });
cv.addEventListener('touchmove', e => { const r = cv.getBoundingClientRect(); aimX = (e.touches[0].clientX - r.left) * cv.width / r.width; });
cv.addEventListener('click', shoot); cv.addEventListener('touchend', shoot);
$('restart').onclick = reset;
function shoot() {
  if (ball || over) return;
  const tx = Math.min(GOAL.x + GOAL.w + 30, Math.max(GOAL.x - 30, aimX));
  ball = { t: 0, tx, curve: +$('curve').value, x: START.x, y: START.y };
}
function finish(msg, scored) {
  if (scored) goals++; shots--; ball = null;
  $('gstat').textContent = `${msg}  Goals: ${goals} / Shots left: ${shots}`;
  if (shots === 0) { over = true; $('gstat').textContent = `Full time! You scored ${goals}/5. ${goals>=4?'Rashford would be proud 🔥':'Press Restart to try again.'}`; }
}
function loop() {
  cx.clearRect(0, 0, cv.width, cv.height);
  cx.fillStyle = '#166534'; cx.fillRect(0, 0, 420, 440);
  cx.strokeStyle = '#fff'; cx.lineWidth = 3; cx.strokeRect(GOAL.x, GOAL.y, GOAL.w, 40);       // goal
  cx.fillStyle = 'rgba(255,255,255,.12)'; cx.fillRect(GOAL.x, GOAL.y, GOAL.w, 40);
  keeper.x += keeper.dir * (ball ? 2.6 : 1.5);
  if (keeper.x > GOAL.x + GOAL.w - 20 || keeper.x < GOAL.x + 20) keeper.dir *= -1;
  cx.fillStyle = '#facc15'; cx.fillRect(keeper.x - 18, GOAL.y + 8, 36, 24);                   // keeper
  cx.fillStyle = '#ef4444';                                                                   // wall
  for (let i = 0; i < 4; i++) { cx.beginPath(); cx.arc(WALL.x + 15 + i * 30, WALL.y + 10, 14, 0, 7); cx.fill(); }
  if (!ball && !over) {                                                                       // aim line
    cx.setLineDash([6, 6]); cx.strokeStyle = 'rgba(255,255,255,.6)'; cx.beginPath();
    cx.moveTo(START.x, START.y); cx.quadraticCurveTo((START.x + aimX) / 2 + (+$('curve').value), 240, aimX, GOAL.y + 20); cx.stroke(); cx.setLineDash([]);
  }
  if (ball) {
    ball.t += 0.016;
    const t = Math.min(ball.t, 1), cxp = (START.x + ball.tx) / 2 + ball.curve, cyp = 240, ey = GOAL.y + 20;
    ball.x = (1-t)*(1-t)*START.x + 2*(1-t)*t*cxp + t*t*ball.tx;
    ball.y = (1-t)*(1-t)*START.y + 2*(1-t)*t*cyp + t*t*ey;
    if (ball.y < WALL.y + WALL.h && ball.y > WALL.y - 10 && ball.x > WALL.x - 5 && ball.x < WALL.x + WALL.w + 5) finish('Blocked by the wall! 🧱', false);
    else if (t >= 1) {
      if (ball.x < GOAL.x || ball.x > GOAL.x + GOAL.w) finish('Wide! 😬', false);
      else if (Math.abs(ball.x - keeper.x) < 24) finish('Saved! 🧤', false);
      else finish('GOAL! ⚽🔥', true);
    }
    if (ball) { cx.fillStyle = '#fff'; cx.beginPath(); cx.arc(ball.x, ball.y, 8 - 3 * ball.t, 0, 7); cx.fill(); }
  } else { cx.fillStyle = '#fff'; cx.beginPath(); cx.arc(START.x, START.y, 8, 0, 7); cx.fill(); }
  requestAnimationFrame(loop);
}

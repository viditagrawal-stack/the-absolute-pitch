const COMPETITIONS = ['PL', 'PD', 'CL']; // stored in competition_name
const TABLE = 'football_live_data';

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const iso = (d) => d.toISOString().slice(0, 10);

async function fd(path) {
  const res = await fetch(`https://api.football-data.org/v4${path}`, {
    headers: { 'X-Auth-Token': FOOTBALL_API_KEY },
  });
  if (!res.ok) throw new Error(`Football-Data ${res.status} on ${path}: ${await res.text()}`);
  return res.json();
}

async function sb(path, method, body) {
  const res = await fetch(`${SUPABASE_URL}/rest/v1/${path}`, {
    method,
    headers: {
      apikey: SUPABASE_KEY,
      Authorization: `Bearer ${SUPABASE_KEY}`,
      'Content-Type': 'application/json',
      Prefer: 'return=representation',
    },
    body: JSON.stringify(body),
  });
  if (!res.ok) throw new Error(`Supabase ${res.status}: ${await res.text()}`);
  return res.json();
}

async function save(code, payload) {
  // PATCH the row if it exists, otherwise POST a new one
  const updated = await sb(`${TABLE}?competition_name=eq.${code}`, 'PATCH', { json_payload: payload });
  if (updated.length === 0) {
    await sb(TABLE, 'POST', { competition_name: code, json_payload: payload });
    console.log(`  inserted new row for ${code}`);
  }
}

function parseStandings(data) {
  const group = (data.standings || []).find((s) => s.type === 'TOTAL') || (data.standings || [])[0];
  return (group?.table || []).map((r) => ({
    position: r.position,
    team: r.team.shortName || r.team.name,
    crest: r.team.crest,
    played: r.playedGames,
    won: r.won,
    draw: r.draw,
    lost: r.lost,
    gf: r.goalsFor,
    ga: r.goalsAgainst,
    gd: r.goalDifference,
    points: r.points,
    form: r.form || '',
  }));
}

function parseMatches(data) {
  return (data.matches || []).map((m) => ({
    id: m.id,
    date: m.utcDate,
    status: m.status, // SCHEDULED, TIMED, IN_PLAY, PAUSED, FINISHED...
    matchday: m.matchday,
    home: m.homeTeam.shortName || m.homeTeam.name,
    away: m.awayTeam.shortName || m.awayTeam.name,
    homeScore: m.score?.fullTime?.home ?? null,
    awayScore: m.score?.fullTime?.away ?? null,
  }));
}

async function updateAll() {
  const from = new Date(Date.now() - 7 * 864e5);
  const to = new Date(Date.now() + 7 * 864e5);
  console.log(`[${new Date().toLocaleTimeString()}] Syncing...`);

  for (const code of COMPETITIONS) {
    try {
      const standingsRaw = await fd(`/competitions/${code}/standings`);
      await sleep(7000); // free tier = 10 requests/min
      const matchesRaw = await fd(`/competitions/${code}/matches?dateFrom=${iso(from)}&dateTo=${iso(to)}`);
      await sleep(7000);

      await save(code, {
        updated: new Date().toISOString(),
        season: standingsRaw.season?.startDate?.slice(0, 4),
        standings: parseStandings(standingsRaw),
        matches: parseMatches(matchesRaw),
      });
      console.log(`  ✓ ${code} saved`);
    } catch (err) {
      console.error(`  ✗ ${code} failed:`, err.message);
    }
  }
}

(async () => {
  await updateAll();
  if (process.argv.includes('--loop')) {
    // ~42s of waiting is already built in per run; loop again after 60s
    while (true) {
      await sleep(60000);
      await updateAll();
    }
  }
})();
